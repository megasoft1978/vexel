   369	            const char *const names_with_name[] = {"id", "kind", "target", "name", "visible", "cels"};
   370	            const char *const names_without_name[] = {"id", "kind", "target", "visible", "cels"};
   371	            if (!only_fields(&track, name_field == NULL ? names_without_name : names_with_name,
   372	                             name_field == NULL ? 5U : 6U) || items_field == NULL)
   373	                return result(PROJECT_TIMELINE_INVALID_SCHEMA, track_spans[ti].start);
   374	        } else {
   375	            const char *const names_with_name[] = {"id", "kind", "target", "name", "visible", "property", "value_type", "interpolation", "keys"};
   376	            const char *const names_without_name[] = {"id", "kind", "target", "visible", "property", "value_type", "interpolation", "keys"};
   377	            if (!only_fields(&track, name_field == NULL ? names_without_name : names_with_name,
   378	                             name_field == NULL ? 8U : 9U)) return result(PROJECT_TIMELINE_INVALID_SCHEMA, track_spans[ti].start);
   379	        }
   380	        size_t item_count = 0U;
   381	        if (items_field == NULL) return result(PROJECT_TIMELINE_INVALID_SCHEMA, track_spans[ti].start);
   382	        base_bytes value_type = {NULL, 0U};
   383	        if (!drawing) {
   384	            const json_field *property_field = get_field(&track, "property");
   385	            const json_field *type_field = get_field(&track, "value_type");
   386	            const json_field *interpolation_field = get_field(&track, "interpolation");
   387	            base_bytes property, interpolation;
   388	            if (property_field == NULL || type_field == NULL || interpolation_field == NULL ||
   389	                !span_string(&reader, property_field->value, &property) || property.len == 0U ||
   390	                !span_string(&reader, type_field->value, &value_type) ||
   391	                !span_string(&reader, interpolation_field->value, &interpolation) ||
   392	                !(json_string_equal(interpolation, "step") || json_string_equal(interpolation, "linear") || json_string_equal(interpolation, "cubic")) ||
   393	                !(json_string_equal(value_type, "scalar") || json_string_equal(value_type, "transform2d") ||
   394	                  json_string_equal(value_type, "rgba") || json_string_equal(value_type, "boolean")))
   395	                return result(PROJECT_TIMELINE_INVALID_VALUE, track_spans[ti].start);
   396	            bool property_matches =
   397	                (json_string_equal(property, "x") || json_string_equal(property, "y") ||
   398	                 json_string_equal(property, "width") || json_string_equal(property, "height") ||
   399	                 json_string_equal(property, "cx") || json_string_equal(property, "cy") ||
   400	                 json_string_equal(property, "r") || json_string_equal(property, "rx") ||
   401	                 json_string_equal(property, "ry") || json_string_equal(property, "opacity") ||
   402	                 json_string_equal(property, "fill-opacity") || json_string_equal(property, "stroke-opacity") ||
   403	                 json_string_equal(property, "stroke-width") || json_string_equal(property, "stroke-dashoffset")) &&
   404	                json_string_equal(value_type, "scalar");
   405	            property_matches = property_matches ||
   406	                ((json_string_equal(property, "fill") || json_string_equal(property, "stroke")) &&
   407	                 json_string_equal(value_type, "rgba")) ||
   408	                (json_string_equal(property, "transform") && json_string_equal(value_type, "transform2d")) ||
   409	                (json_string_equal(property, "visibility") && json_string_equal(value_type, "boolean"));
   410	            if (!property_matches) return result(PROJECT_TIMELINE_INVALID_VALUE, property_field->value.start);
   411	            if (json_string_equal(value_type, "boolean") && !json_string_equal(interpolation, "step"))
   412	                return result(PROJECT_TIMELINE_INVALID_VALUE, interpolation_field->value.start);
   413	        }
   414	        item_validation item_state = {drawing, value_type, duration, 0U, context};
   415	        if (!array_visit(&reader, items_field->value, PROJECT_TIMELINE_MAX_ITEMS_PER_TRACK,
   416	                         validate_item_visit, &item_state, &item_count))
   417	            return result(reader.error_kind, reader.error);
   418	        if (item_count == 0U) return result(PROJECT_TIMELINE_INVALID_VALUE, items_field->value.start);
   419	        if (!base_size_add(total_items, item_count, &total_items) || total_items > PROJECT_TIMELINE_MAX_ITEMS)
   420	            return result(PROJECT_TIMELINE_LIMIT, items_field->value.start);
   421	    }
   422	    project_timeline_view decoded = {0};
   423	    decoded.source = source;
   424	    decoded.tracks_offset = f_tracks->value.start;
   425	    decoded.tracks_length = f_tracks->value.end - f_tracks->value.start;
   426	    decoded.ticks_per_second = tps;
   427	    decoded.duration_ticks = duration;
   428	    decoded.fps_numerator = numerator;
   429	    decoded.fps_denominator = denominator;
   430	    decoded.track_count = track_count;
   431	    for (size_t i = 0U; i < track_count; i++) {
   432	        decoded.track_spans[i] = (base_bytes){source.ptr + track_spans[i].start,
   433	            track_spans[i].end - track_spans[i].start};
   434	        object_fields fields;
   435	        if (!parse_object_fields(&reader, track_spans[i], &fields))
   436	            return result(reader.error_kind, reader.error);
   437	        const json_field *kind_field = get_field(&fields, "kind");
   438	        const json_field *items_field = NULL;
   439	        base_bytes kind;
   440	        if (kind_field == NULL || !span_string(&reader, kind_field->value, &kind))
   441	            return result(PROJECT_TIMELINE_INVALID_SCHEMA, track_spans[i].start);
   442	        decoded.drawing_tracks[i] = json_string_equal(kind, "drawing");
   443	        items_field = get_field(&fields, decoded.drawing_tracks[i] ? "cels" : "keys");
   444	        if (items_field == NULL) return result(PROJECT_TIMELINE_INVALID_SCHEMA, track_spans[i].start);
   445	        decoded.item_spans[i] = (base_bytes){source.ptr + items_field->value.start,
   446	            items_field->value.end - items_field->value.start};
   447	        size_t item_count = 0U;
   448	        if (!array_count(&reader, items_field->value, PROJECT_TIMELINE_MAX_ITEMS_PER_TRACK,
   449	                         &item_count)) return result(reader.error_kind, reader.error);
   450	        decoded.item_counts[i] = item_count;
   451	    }
   452	    *out_view = decoded;
   453	    return result(PROJECT_TIMELINE_OK, 0U);
   454	}

   516	bool project_timeline_item_next(const project_timeline_view *view, size_t track_index,
   517	                               project_timeline_iterator *iterator,
   518	                               project_timeline_item_view *out_item)
   519	{
   520	    if (view == NULL || iterator == NULL || out_item == NULL || track_index >= view->track_count ||
   521	        view->track_count > PROJECT_TIMELINE_MAX_TRACKS ||
   522	        !base_bytes_valid(view->item_spans[track_index])) return false;
   523	    if (iterator->initialized && iterator->track_index != track_index) return false;
   524	    size_t array_start = (size_t)(view->item_spans[track_index].ptr - view->source.ptr);
   525	    size_t array_end = array_start + view->item_spans[track_index].len;
   526	    if (!iterator->initialized) {
   527	        iterator->track_index = track_index;
   528	        iterator->next_index = 0U;
   529	        iterator->source_cursor = array_start;
   530	        iterator->initialized = true;
   531	    }
   532	    if (iterator->source_cursor < array_start || iterator->source_cursor > array_end ||
   533	        iterator->source_cursor == SIZE_MAX) return false;
   534	    json_reader reader = {view->source, iterator->source_cursor, 0U, PROJECT_TIMELINE_OK};
   535	    json_span item_span;
   536	    if (iterator->next_index == 0U) {
   537	        reader.cursor = array_start;
   538	        skip_space(&reader);
   539	        if (reader.cursor >= reader.source.len || reader.source.ptr[reader.cursor++] != '[') return false;
   540	        skip_space(&reader);
   541	    }
   542	    size_t start = reader.cursor;
   543	    if (reader.cursor >= reader.source.len || reader.source.ptr[reader.cursor] == ']') return false;
   544	    if (!scan_value(&reader, 1U)) return false;
   545	    item_span = (json_span){start, reader.cursor};
   546	    skip_space(&reader);
   547	    if (reader.cursor >= reader.source.len) return false;
   548	    uint8_t delimiter = reader.source.ptr[reader.cursor++];
   549	    if (delimiter != ',' && delimiter != ']') return false;
   550	    if (delimiter == ',') skip_space(&reader);
   551	    iterator->source_cursor = delimiter == ']' ? SIZE_MAX : reader.cursor;
   552	    object_fields item;
   553	    if (!parse_object_fields(&reader, item_span, &item)) return false;
   554	    const json_field *tick_field = get_field(&item, "tick");
   555	    uint32_t tick = 0U;
   556	    if (tick_field == NULL || !span_uint(&reader, tick_field->value, &tick)) return false;
   557	    project_timeline_item_view output = {tick, {NULL, 0U}, {NULL, 0U},
   558	        item_span.start, item_span.end - item_span.start};
   559	    if (view->drawing_tracks[track_index]) {
   560	        const json_field *asset = get_field(&item, "asset");
   561	        if (asset == NULL || !span_string(&reader, asset->value, &output.asset)) return false;
   562	    } else {
   563	        const json_field *value = get_field(&item, "value");
   564	        if (value == NULL) return false;
   565	        output.value_json = (base_bytes){view->source.ptr + value->value.start,
   566	            value->value.end - value->value.start};
   567	    }
   568	    iterator->next_index++;
   569	    *out_item = output;
   570	    return true;
   571	}

    46	static bool utf8_sequence(const uint8_t *bytes, size_t length, size_t *cursor)
    47	{
    48	    uint8_t first = bytes[*cursor];
    49	    size_t width = first >= 0xc2U && first <= 0xdfU ? 2U :
    50	        first >= 0xe0U && first <= 0xefU ? 3U :
    51	        first >= 0xf0U && first <= 0xf4U ? 4U : 0U;
    52	    if (width == 0U || width > length - *cursor) return false;
    53	    uint8_t second = bytes[*cursor + 1U];
    54	    if ((second & 0xc0U) != 0x80U ||
    55	        (first == 0xe0U && second < 0xa0U) ||
    56	        (first == 0xedU && second >= 0xa0U) ||
    57	        (first == 0xf0U && second < 0x90U) ||
    58	        (first == 0xf4U && second >= 0x90U)) return false;
    59	    for (size_t i = 2U; i < width; i++)
    60	        if ((bytes[*cursor + i] & 0xc0U) != 0x80U) return false;
    61	    *cursor += width;
    62	    return true;
    63	}

    65	static int hex_value(uint8_t byte)
    66	{
    67	    if (byte >= '0' && byte <= '9') return (int)(byte - '0');
    68	    if (byte >= 'a' && byte <= 'f') return (int)(byte - 'a') + 10;
    69	    if (byte >= 'A' && byte <= 'F') return (int)(byte - 'A') + 10;
    70	    return -1;
    71	}

    73	static bool append_utf8(uint32_t codepoint, base_mut_bytes output, size_t *length)
    74	{
    75	    uint8_t bytes[4];
    76	    size_t count;
    77	    if (codepoint <= 0x7fU) { bytes[0] = (uint8_t)codepoint; count = 1U; }
    78	    else if (codepoint <= 0x7ffU) {
    79	        bytes[0] = (uint8_t)(0xc0U | (codepoint >> 6U));
    80	        bytes[1] = (uint8_t)(0x80U | (codepoint & 0x3fU)); count = 2U;
    81	    } else if (codepoint <= 0xffffU) {
    82	        bytes[0] = (uint8_t)(0xe0U | (codepoint >> 12U));
    83	        bytes[1] = (uint8_t)(0x80U | ((codepoint >> 6U) & 0x3fU));
    84	        bytes[2] = (uint8_t)(0x80U | (codepoint & 0x3fU)); count = 3U;
    85	    } else if (codepoint <= 0x10ffffU) {
    86	        bytes[0] = (uint8_t)(0xf0U | (codepoint >> 18U));
    87	        bytes[1] = (uint8_t)(0x80U | ((codepoint >> 12U) & 0x3fU));
    88	        bytes[2] = (uint8_t)(0x80U | ((codepoint >> 6U) & 0x3fU));
    89	        bytes[3] = (uint8_t)(0x80U | (codepoint & 0x3fU)); count = 4U;
    90	    } else return false;
    91	    if (*length > output.len || count > output.len - *length) return false;
    92	    memcpy(output.ptr + *length, bytes, count);
    93	    *length += count;
    94	    return true;
    95	}

    97	static bool decode_string_content(base_bytes encoded, base_mut_bytes output,
    98	                                  size_t *out_length, bool *out_limit)
    99	{
   100	    if (!base_bytes_valid(encoded) || (output.ptr == NULL && output.len != 0U) || out_length == NULL)
   101	        return false;
   102	    size_t input = 0U, length = 0U;
   103	    if (out_limit != NULL) *out_limit = false;
   104	    while (input < encoded.len) {
   105	        uint8_t byte = encoded.ptr[input++];
   106	        uint32_t codepoint = byte;
   107	        if (byte == '\\') {
   108	            if (input >= encoded.len) return false;
   109	            uint8_t escaped = encoded.ptr[input++];
   110	            if (escaped == '"' || escaped == '\\' || escaped == '/') codepoint = escaped;
   111	            else if (escaped == 'b') codepoint = 8U;
   112	            else if (escaped == 'f') codepoint = 12U;
   113	            else if (escaped == 'n') codepoint = 10U;
   114	            else if (escaped == 'r') codepoint = 13U;
   115	            else if (escaped == 't') codepoint = 9U;
   116	            else if (escaped == 'u') {
   117	                if (encoded.len - input < 4U) return false;
   118	                codepoint = 0U;
   119	                for (size_t i = 0U; i < 4U; i++) {
   120	                    int digit = hex_value(encoded.ptr[input++]);
   121	                    if (digit < 0) return false;
   122	                    codepoint = (codepoint << 4U) | (uint32_t)digit;
   123	                }
   124	                if (codepoint >= 0xd800U && codepoint <= 0xdbffU) {
   125	                    if (encoded.len - input < 6U || encoded.ptr[input++] != '\\' ||
   126	                        encoded.ptr[input++] != 'u') return false;
   127	                    uint32_t low = 0U;
   128	                    for (size_t i = 0U; i < 4U; i++) {
   129	                        int digit = hex_value(encoded.ptr[input++]);
   130	                        if (digit < 0) return false;
   131	                        low = (low << 4U) | (uint32_t)digit;
   132	                    }
   133	                    if (low < 0xdc00U || low > 0xdfffU) return false;
   134	                    codepoint = 0x10000U + ((codepoint - 0xd800U) << 10U) + (low - 0xdc00U);
   135	                } else if (codepoint >= 0xdc00U && codepoint <= 0xdfffU) return false;
   136	            } else return false;
   137	        } else if (byte >= 0x80U) {
   138	            input--;
   139	            size_t start = input;
   140	            if (!utf8_sequence(encoded.ptr, encoded.len, &input)) return false;
   141	            size_t width = input - start;
   142	            if (width > output.len - length) {
   143	                if (out_limit != NULL) *out_limit = true;
   144	                return false;
   145	            }
   146	            memcpy(output.ptr + length, encoded.ptr + start, width);
   147	            length += width;
   148	            continue;
   149	        } else if (byte < 0x20U) return false;
   150	        if (!append_utf8(codepoint, output, &length)) {
   151	            if (out_limit != NULL) *out_limit = true;
   152	            return false;
   153	        }
   154	        if (length > PROJECT_TIMELINE_MAX_STRING_BYTES) {
   155	            if (out_limit != NULL) *out_limit = true;
   156	            return false;
   157	        }
   158	    }
   159	    *out_length = length;
   160	    return true;
   161	}

   163	static bool scan_string(json_reader *reader, base_bytes *out)
   164	{
   165	    skip_space(reader);
   166	    if (reader->cursor >= reader->source.len ||
   167	        reader->source.ptr[reader->cursor] != '"') return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   168	    size_t start = ++reader->cursor;
   169	    while (reader->cursor < reader->source.len) {
   170	        uint8_t byte = reader->source.ptr[reader->cursor];
   171	        if (byte == '"') {
   172	            size_t end = reader->cursor++;
   173	            base_bytes raw = {reader->source.ptr + start, end - start};
   174	            char decoded[PROJECT_TIMELINE_MAX_STRING_BYTES];
   175	            size_t decoded_length = 0U;
   176	            bool limited = false;
   177	            if (!decode_string_content(raw,
   178	                    (base_mut_bytes){(uint8_t *)decoded, sizeof(decoded)},
   179	                    &decoded_length, &limited))
   180	                return fail(reader, limited ? PROJECT_TIMELINE_LIMIT : PROJECT_TIMELINE_MALFORMED_JSON);
   181	            if (out != NULL) *out = raw;
   182	            return true;
   183	        }
   184	        if (byte < 0x20U) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   185	        if (byte == '\\') {
   186	            reader->cursor++;
   187	            if (reader->cursor >= reader->source.len) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   188	            uint8_t escaped = reader->source.ptr[reader->cursor++];
   189	            if (escaped == '"' || escaped == '\\' || escaped == '/' || escaped == 'b' ||
   190	                escaped == 'f' || escaped == 'n' || escaped == 'r' || escaped == 't') {
   191	                continue;
   192	            }
   193	            if (escaped != 'u' || reader->source.len - reader->cursor < 4U)
   194	                return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   195	            unsigned code = 0U;
   196	            for (size_t i = 0U; i < 4U; i++) {
   197	                int digit = hex_value(reader->source.ptr[reader->cursor++]);
   198	                if (digit < 0) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   199	                code = (code << 4U) | (unsigned)digit;
   200	            }
   201	            if (code >= 0xd800U && code <= 0xdbffU) {
   202	                if (reader->source.len - reader->cursor < 6U ||
   203	                    reader->source.ptr[reader->cursor] != '\\' ||
   204	                    reader->source.ptr[reader->cursor + 1U] != 'u')
   205	                    return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   206	                reader->cursor += 2U;
   207	                uint32_t low = 0U;
   208	                for (size_t i = 0U; i < 4U; i++) {
   209	                    int digit = hex_value(reader->source.ptr[reader->cursor++]);
   210	                    if (digit < 0) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   211	                    low = (low << 4U) | (uint32_t)digit;
   212	                }
   213	                if (low < 0xdc00U || low > 0xdfffU)
   214	                    return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   215	            } else if (code >= 0xdc00U && code <= 0xdfffU) {
   216	                return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   217	            }
   218	            continue;
   219	        }
   220	        if (byte >= 0x80U) {
   221	            if (!utf8_sequence(reader->source.ptr, reader->source.len, &reader->cursor))
   222	                return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   223	        } else reader->cursor++;
   224	    }
   225	    return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   226	}

   228	bool scan_number(json_reader *reader)
   229	{
   230	    size_t start = reader->cursor;
   231	    if (reader->source.ptr[reader->cursor] == '-') reader->cursor++;
   232	    if (reader->cursor >= reader->source.len) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   233	    if (reader->source.ptr[reader->cursor] == '0') reader->cursor++;
   234	    else {
   235	        if (reader->source.ptr[reader->cursor] < '1' || reader->source.ptr[reader->cursor] > '9')
   236	            return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   237	        while (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] >= '0' &&
   238	               reader->source.ptr[reader->cursor] <= '9') reader->cursor++;
   239	    }
   240	    if (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] == '.') {
   241	        reader->cursor++;
   242	        size_t digits = reader->cursor;
   243	        while (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] >= '0' &&
   244	               reader->source.ptr[reader->cursor] <= '9') reader->cursor++;
   245	        if (reader->cursor == digits) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   246	    }
   247	    if (reader->cursor < reader->source.len &&
   248	        (reader->source.ptr[reader->cursor] == 'e' || reader->source.ptr[reader->cursor] == 'E')) {
   249	        reader->cursor++;
   250	        if (reader->cursor < reader->source.len &&
   251	            (reader->source.ptr[reader->cursor] == '+' || reader->source.ptr[reader->cursor] == '-'))
   252	            reader->cursor++;
   253	        size_t digits = reader->cursor;
   254	        while (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] >= '0' &&
   255	               reader->source.ptr[reader->cursor] <= '9') reader->cursor++;
   256	        if (reader->cursor == digits) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   257	    }
   258	    if (reader->cursor - start > 96U) return fail(reader, PROJECT_TIMELINE_LIMIT);
   259	    char buffer[97];
   260	    size_t length = reader->cursor - start;
   261	    memcpy(buffer, reader->source.ptr + start, length);
   262	    buffer[length] = '\0';
   263	    char *end = NULL;
   264	    (void)pthread_once(&timeline_locale_once, initialize_timeline_locale);
   265	    if (timeline_c_locale == (locale_t)0) return fail(reader, PROJECT_TIMELINE_INVALID_VALUE);
   266	    double value = strtod_l(buffer, &end, timeline_c_locale);
   267	    if (end != buffer + length || !isfinite(value)) return fail(reader, PROJECT_TIMELINE_INVALID_VALUE);
   268	    return true;
   269	}

   273	static bool scan_array(json_reader *reader, unsigned depth)
   274	{
   275	    reader->cursor++;
   276	    skip_space(reader);
   277	    if (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] == ']') {
   278	        reader->cursor++;
   279	        return true;
   280	    }
   281	    for (;;) {
   282	        if (!scan_value(reader, depth + 1U)) return false;
   283	        skip_space(reader);
   284	        if (reader->cursor >= reader->source.len) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   285	        uint8_t byte = reader->source.ptr[reader->cursor++];
   286	        if (byte == ']') return true;
   287	        if (byte != ',') return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   288	        skip_space(reader);
   289	    }
   290	}

   292	static bool scan_object(json_reader *reader, unsigned depth)
   293	{
   294	    reader->cursor++;
   295	    base_bytes keys[TIMELINE_MAX_FIELDS];
   296	    size_t key_count = 0U;
   297	    skip_space(reader);
   298	    if (reader->cursor < reader->source.len && reader->source.ptr[reader->cursor] == '}') {
   299	        reader->cursor++;
   300	        *reader = (json_reader){reader->source, reader->cursor, reader->error, reader->error_kind};
   301	        return true;
   302	    }
   303	    for (;;) {
   304	        size_t key_offset = reader->cursor;
   305	        base_bytes key;
   306	        if (!scan_string(reader, &key)) return false;
   307	        for (size_t i = 0U; i < key_count; i++) {
   308	            if (json_strings_equal(keys[i], key)) {
   309	                reader->cursor = key_offset;
   310	                return fail(reader, PROJECT_TIMELINE_DUPLICATE);
   311	            }
   312	        }
   313	        if (key_count >= TIMELINE_MAX_FIELDS) return fail(reader, PROJECT_TIMELINE_LIMIT);
   314	        keys[key_count++] = key;
   315	        skip_space(reader);
   316	        if (reader->cursor >= reader->source.len || reader->source.ptr[reader->cursor++] != ':')
   317	            return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   318	        if (!scan_value(reader, depth + 1U)) return false;
   319	        skip_space(reader);
   320	        if (reader->cursor >= reader->source.len) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   321	        uint8_t byte = reader->source.ptr[reader->cursor++];
   322	        if (byte == '}') return true;
   323	        if (byte != ',') return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   324	        skip_space(reader);
   325	    }
   326	}

   328	bool scan_value(json_reader *reader, unsigned depth)
   329	{
   330	    if (depth > 32U) return fail(reader, PROJECT_TIMELINE_LIMIT);
   331	    skip_space(reader);
   332	    if (reader->cursor >= reader->source.len) return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   333	    uint8_t byte = reader->source.ptr[reader->cursor];
   334	    if (byte == '"') return scan_string(reader, NULL);
   335	    if (byte == '{') return scan_object(reader, depth);
   336	    if (byte == '[') return scan_array(reader, depth);
   337	    if (byte == '-' || (byte >= '0' && byte <= '9')) return scan_number(reader);
   338	    static const char *const literals[] = {"true", "false", "null"};
   339	    for (size_t i = 0U; i < sizeof(literals) / sizeof(literals[0]); i++) {
   340	        size_t n = strlen(literals[i]);
   341	        if (reader->source.len - reader->cursor >= n &&
   342	            memcmp(reader->source.ptr + reader->cursor, literals[i], n) == 0) {
   343	            reader->cursor += n;
   344	            return true;
   345	        }
   346	    }
   347	    return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   348	}

   351	{
   352	    return a.len == b.len && (a.len == 0U || memcmp(a.ptr, b.ptr, a.len) == 0);

   355	int bytes_compare(base_bytes a, base_bytes b)
   356	{
   357	    size_t common = a.len < b.len ? a.len : b.len;
   358	    int order = common == 0U ? 0 : memcmp(a.ptr, b.ptr, common);
   359	    if (order != 0) return order;
   360	    return a.len < b.len ? -1 : a.len > b.len ? 1 : 0;
   361	}

   391	bool parse_object_fields(json_reader *reader, json_span span,
   392	                                object_fields *out)
   393	{
   394	    json_reader child = *reader;
   395	    child.cursor = span.start;
   396	    skip_space(&child);
   397	    if (child.cursor >= child.source.len || child.source.ptr[child.cursor++] != '{') {
   398	        *reader = child;
   399	        return fail_at(reader, PROJECT_TIMELINE_INVALID_SCHEMA, span.start);
   400	    }
   401	    out->count = 0U;
   402	    skip_space(&child);
   403	    if (child.cursor < child.source.len && child.source.ptr[child.cursor] == '}') {
   404	        child.cursor++;
   405	        *reader = child;
   406	        return true;
   407	    }
   408	    for (;;) {
   409	        base_bytes encoded_key;
   410	        if (!scan_string(&child, &encoded_key)) { *reader = child; return false; }
   411	        char decoded_key[128];
   412	        size_t decoded_key_length = 0U;
   413	        if (!project_timeline_string_copy(encoded_key,
   414	                (base_mut_bytes){(uint8_t *)decoded_key, sizeof(decoded_key) - 1U},
   415	                &decoded_key_length)) {
   416	            *reader = child; return fail(reader, PROJECT_TIMELINE_LIMIT);
   417	        }
   418	        decoded_key[decoded_key_length] = '\0';
   419	        base_bytes key = {(const uint8_t *)decoded_key, decoded_key_length};
   420	        for (size_t i = 0U; i < out->count; i++)
   421	            if (bytes_equal(out->fields[i].key, key)) {
   422	                *reader = child; return fail(reader, PROJECT_TIMELINE_DUPLICATE);
   423	            }
   424	        skip_space(&child);
   425	        if (child.cursor >= child.source.len || child.source.ptr[child.cursor++] != ':') {
   426	            *reader = child; return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON);
   427	        }
   428	        skip_space(&child);
   429	        if (out->count >= TIMELINE_MAX_FIELDS) { *reader = child; return fail(reader, PROJECT_TIMELINE_LIMIT); }
   430	        size_t start = child.cursor;
   431	        if (!scan_value(&child, 1U)) { *reader = child; return false; }
   432	        json_field *field = &out->fields[out->count++];
   433	        memcpy(field->key_storage, decoded_key, decoded_key_length + 1U);
   434	        field->key = (base_bytes){(const uint8_t *)field->key_storage, decoded_key_length};
   435	        field->value = (json_span){start, child.cursor};
   436	        skip_space(&child);
   437	        if (child.cursor >= child.source.len) { *reader = child; return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON); }
   438	        uint8_t byte = child.source.ptr[child.cursor++];
   439	        if (byte == '}') { *reader = child; return true; }
   440	        if (byte != ',') { *reader = child; return fail(reader, PROJECT_TIMELINE_MALFORMED_JSON); }
   441	        skip_space(&child);
   442	    }
   443	}

   604	bool array_item_at_from(json_reader *reader, json_span span, size_t wanted,
   605	                        size_t *in_out_cursor, size_t *in_out_index,
   606	                        json_span *out_item)
   607	{
   608	    if (reader == NULL || in_out_cursor == NULL || in_out_index == NULL ||
   609	        out_item == NULL || wanted < *in_out_index || *in_out_cursor > span.end)
   610	        return false;
   611	    json_reader child = *reader;
   612	    child.cursor = *in_out_cursor;
   613	    size_t index = *in_out_index;
   614	    if (index == 0U && child.cursor == 0U) {
   615	        child.cursor = span.start;
   616	        skip_space(&child);
   617	        if (child.cursor >= child.source.len || child.source.ptr[child.cursor++] != '[')
   618	            return false;
   619	        skip_space(&child);
   620	    }
   621	    while (index <= wanted) {
   622	        if (child.cursor >= span.end || child.source.ptr[child.cursor] == ']') return false;
   623	        size_t start = child.cursor;
   624	        if (!scan_value(&child, 1U)) { *reader = child; return false; }
   625	        size_t end = child.cursor;
   626	        skip_space(&child);

    73	static bool append_utf8(uint32_t codepoint, base_mut_bytes output, size_t *length)
    74	{
    75	    uint8_t bytes[4];
    76	    size_t count;
    77	    if (codepoint <= 0x7fU) { bytes[0] = (uint8_t)codepoint; count = 1U; }
    78	    else if (codepoint <= 0x7ffU) {
    79	        bytes[0] = (uint8_t)(0xc0U | (codepoint >> 6U));
    80	        bytes[1] = (uint8_t)(0x80U | (codepoint & 0x3fU)); count = 2U;
    81	    } else if (codepoint <= 0xffffU) {
    82	        bytes[0] = (uint8_t)(0xe0U | (codepoint >> 12U));
    83	        bytes[1] = (uint8_t)(0x80U | ((codepoint >> 6U) & 0x3fU));
    84	        bytes[2] = (uint8_t)(0x80U | (codepoint & 0x3fU)); count = 3U;

    86	        bytes[0] = (uint8_t)(0xf0U | (codepoint >> 18U));
    87	        bytes[1] = (uint8_t)(0x80U | ((codepoint >> 12U) & 0x3fU));
    88	        bytes[2] = (uint8_t)(0x80U | ((codepoint >> 6U) & 0x3fU));
