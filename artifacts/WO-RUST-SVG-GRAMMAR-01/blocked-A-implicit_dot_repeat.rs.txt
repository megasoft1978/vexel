use vexel_geometry::affine_identity;
use vexel_svg_path::{parse_path, ParseOptions};
use vexel_base::Span;

fn values(source: &[u8]) -> Result<Vec<f64>, vexel_svg_path::Diagnostic> {
    parse_path(
        source,
        Span {
            start: 0,
            end: source.len(),
        },
        affine_identity(),
        ParseOptions::default(),
    )
    .map(|path| path.numbers.iter().map(|n| n.value).collect())
}

#[test]
fn compact_dot_implicit_repeat_matches_spaced_form() {
    for (compact, spaced) in [
        (&b"M0 0l-.1.21.1.55z"[..], &b"M0 0l-.1 .21 .1 .55z"[..]),
        (&b"M0 0 L1 2.5.6 7"[..], &b"M0 0 L1 2.5 .6 7"[..]),
    ] {
        assert_eq!(values(compact), values(spaced), "{compact:?}");
    }
}
